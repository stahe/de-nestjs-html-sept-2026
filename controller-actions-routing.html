import { AppModule } from './app.module';

async function bootstrap() {

    Kommentieren wir diesen Code:

    • Zeile 4: Die Startfunktion ist asynchron ([async]): Das Erstellen der Anwendung und das Starten des Servers brauchen Zeit, und wir warten ([await]) auf den Abschluss jedes Schritts;
    • Zeile 6: erstellt die Anwendung aus ihrem Wurzelmodul. [NestJS] findet dann alle Controller und Services der Anwendung und erzeugt ihre Instanzen;

    import { Controller, Get, Param, Post } from '@nestjs/common';

@Controller('first')
export class FirstController {
  // GET /first
  @Get()
  index(): string {
    return 'Action [index] of the [first] controller';
  }

  // GET /first/action01
  @Get('action01')
  action01(): string {
    return 'Action [action01] of the [first] controller, GET request';
  }

  // POST /first/action01: gleicher Pfad, andere HTTP-Methode
  @Post('action01')
  action01Post(): string {
    return 'Action [action01Post] of the [first] controller, POST request';
  }

  // GET /first/products/17: [:id] ist ein Routenparameter
  @Get('products/:id')
  product(@Param('id') id: string): string {
    return `Product no. ${id}`;
  }

  // GET /first/products/17/reviews/3: zwei Routenparameter
  @Get('products/:id/reviews/:number')
  review(@Param() params: { id: string; number: string }): string {
    return `Review no. ${params.number} of product no. ${params.id}`;
  }

  // GET /first/files/...: [*] akzeptiert jedes beliebige Pfadende
  @Get('files/*')
  file(@Param('0') path: string): string {
    return `Requested file: ${path}`;
  }
}

    Gehen wir diesen Code im Einzelnen durch:

    • Zeile 3: [@Controller('first')] legt das Präfix fest, das allen Routen des Controllers gemeinsam ist: Seine Actions bearbeiten URL-Routen, die mit [/first] beginnen;
    • Zeilen 6–7: [@Get()] ohne Argumente: Die Action [index] bearbeitet [GET /first];
    • Zeilen 12–13: Die Action [action01] bearbeitet [GET /first/action01]. Die URL ist die Verkettung aus dem Controller-Präfix und dem Pfad der Action;
    • Zeilen 18–19: der gleiche Pfad, aber für die Methode POST: [@Post] ersetzt [@Get]. Es gibt außerdem [@Put], [@Delete], [@Patch], [@All] ... Der [TypeScript] -Methodenname ([action01Post]) ist beliebig: Für das Routing zählen nur die Dekoratoren;
    • Zeilen 24–25: [:id] ist ein Routenparameter: Dieses URL-Segment kann jeden beliebigen Wert annehmen. Der Dekorator [@Param('id')] injiziert seinen Wert in den Parameter [id] der Action. Dieser Wert ist immer ein String: Für [/first/products/17] ist [id] gleich ['17'], nicht die Zahl 17;
    • Zeilen 30–31: zwei Routenparameter. [@Param()] ohne Argument injiziert ein Objekt, das alle enthält: [{ id: '17', number: '3' }];
    • Zeilen 36–37: Das Zeichen [*] akzeptiert jedes beliebige URL-Ende, einschließlich [/]. Der entsprechende Teil ist der Parameter mit dem Namen [0].

    import { Controller, Get, Header, HttpCode, NotFoundException } from '@nestjs/common';

@Controller('responses')
export class ResponsesController {
  // ein String: wird unverändert gesendet, mit dem Typ [text/html]
  @Get('html')
  html(): string {
    return '<h1>Action [html]</h1>';
  }

  // derselbe String, aber als reiner Text deklariert
  @Get('text')
  @Header('Content-Type', 'text/plain; charset=utf-8')
  text(): string {
    return '<h1>Action [text]</h1>';
  }

  // ein Objekt: wird in JSON serialisiert, mit dem Typ [application/json]
  @Get('json')
  json(): object {
    return { lastName: 'Martin', firstName: 'Jean', age: 42 };
  }

  // ein Array: wird ebenfalls in JSON serialisiert
  @Get('array')
  array(): string[] {
    return ['Mme', 'Mlle', 'Mr'];
  }

  // ein gewählter Statuscode: 201 (Created) statt 200 (OK)
  @Get('created')
  @HttpCode(201)
  created(): object {
    return { message: 'resource created' };
  }

  // keine Antwort: der Code 204 (No Content) bedeutet „nichts anzuzeigen"
  @Get('empty')
  @HttpCode(204)
  empty(): void {}

  // eine asynchrone Methode: NestJS wartet, bis das Promise erfüllt ist
  @Get('promise')
  async promise(): Promise<object> {
    const start = Date.now();
    await new Promise((done) => setTimeout(done, 500));
    return { wait: `${Date.now() - start} ms` };
  }

  // eine Exception: NestJS wandelt sie in eine Fehlerantwort um
  @Get('error')
  error(): string {
    throw new NotFoundException('This resource does not exist');
  }
}

    Gehen wir diesen Code im Einzelnen durch:

    • Zeilen 6–9: Ein String wird unverändert gesendet, mit dem Header [Content-Type: text/html; charset=utf-8];
    • Zeilen 12–16: Der Dekorator [@Header] setzt einen Antwort-Header. Wird er als einfacher Text deklariert ([text/plain]), wird derselbe String vom Browser nicht mehr interpretiert;
    • Zeilen 19–22: Ein Objekt wird als JSON serialisiert, mit dem Header [Content-Type: application/json];
    • Zeilen 25–28: Auch ein Array wird als JSON serialisiert;
    • Zeilen 31–35: [@HttpCode(201)] ändert den Statuscode der Antwort (Standard ist 200 bei einem GET);
    • Zeilen 38–40: Der Code 204 (No Content) bedeutet „kein Inhalt“: Die Action gibt nichts zurück;
    • Zeilen 43–48: Eine Action kann asynchron ([async]) sein: In diesem Fall gibt sie ein Promise ([Promise]) zurück, und [NestJS] wartet auf dessen Auflösung, bevor es antwortet. Das ist bei allen Actions der Fall, die eine Datenbank abfragen;
    • Zeilen 51–54: Eine Action kann eine Exception auslösen. Die Exceptions von [NestJS] ([NotFoundException], [BadRequestException], [ForbiddenException] ...) tragen jeweils einen HTTP-Statuscode (hier 404), den [NestJS] für seine Antwort verwendet.

    import { Controller, Get, Param, Redirect } from '@nestjs/common';

@Controller()
export class RedirectsController {
  // feste Weiterleitung: Adresse und Code stehen im Voraus fest
  @Get('old')
  @Redirect('/new', 301)
  oldPage(): void {}

  @Get('new')
  newPage(): string {
    return 'You are on the [new] page';
  }

  // berechnete Weiterleitung: das zurückgegebene Objekt ersetzt die Adresse von @Redirect()
  @Get('go/:destination')
  @Redirect()
  go(@Param('destination') destination: string): { url: string } {
    return { url: `/page/${destination}` };
  }

  @Get('page/:name')
  page(@Param('name') name: string): string {
    return `You are on the [${name}] page`;
  }

  // Weiterleitung auf eine andere Website
  @Get('documentation')
  @Redirect('https://docs.nestjs.com')
  documentation(): void {}
}

    Gehen wir diesen Code im Einzelnen durch:

    • Zeilen 6–8: [@Redirect('/new', 301)]: eine dauerhafte Weiterleitung. Der Code 301 (Moved Permanently) zeigt an, dass sich die Adresse der Seite dauerhaft geändert hat; der Standardcode ist 302 (Found), eine temporäre Weiterleitung;
    • Zeilen 16–20: eine berechnete Weiterleitung: [@Redirect()] ohne Argument, und die Action gibt ein Objekt [{ url: ... }] zurück, das die Zieladresse festlegt;
    • Zeilen 28–30: Sie können auch auf eine andere Website weiterleiten.

    2.5. Beispiel 05: Die Objekte Request und Response in Express

    [NestJS] baut auf dem Webserver [Express] auf. Eine Action kann direkt auf die [Request] - und [Response] -Objekte von [Express] zugreifen.     import { Controller, Get, Req, Res } from '@nestjs/common';
import { Request, Response } from 'express';

@Controller('express')
export class ExpressController {
  // die Express-Anfrage: Methode, URL, Header ...
  @Get('request')
  request(@Req() req: Request): object {
    return {
      method: req.method,
      url: req.url,
      path: req.path,
      browser: req.get('user-agent'),
    };
  }

  // die Express-Antwort: die Action sendet die Antwort selbst
  @Get('response')
  response(@Res() res: Response): void {
    res.status(200).type('text/plain').send('Response sent by [res.send]');
  }

  // Weiterleitung mit der Express-Antwort
  @Get('redirect')
  redirect(@Res() res: Response): void {
    res.redirect('/express/response');
  }

  // Passthrough: wir verwenden die Antwort (hier für einen Header), aber NestJS
  // sendet weiterhin den von der Action zurückgegebenen Wert
  @Get('passthrough')
  passthrough(@Res({ passthrough: true }) res: Response): string {
    res.setHeader('X-Example', '05-express');
    return 'Response sent by NestJS, with an added header';
  }

  // FEHLER, den man nicht machen sollte: @Res() ohne Passthrough, und nichts wird gesendet
  @Get('blocked')
  blocked(@Res() res: Response): string {
    return 'This string will never be sent: the browser keeps waiting...';
  }
}

    Gehen wir diesen Code im Einzelnen durch:

    • Zeile 2: die Typen [Request] und [Response] von [Express];
    • Zeilen 7–15: [@Req()] injiziert die [Express]-Anfrage: HTTP-Methode, URL, Pfad, Header ([req.get(...)]) ... ;
    • Zeilen 18–21: [@Res()] injiziert die [Express]-Antwort. Hinweis: Sobald eine Action [@Res()] verwendet, [NestJS] geht es davon aus, dass sie selbst die Antwort sendet, und ignoriert ihren Rückgabewert. Hier setzt die Action Statuscode und Typ und sendet den Rumpf mit [res.send];
    • Zeilen 24–27: eine 302-Weiterleitung, die von [Express] mit [res.redirect(url)] ausgeführt wird;
    • Zeilen 31–35: Mit der Option [passthrough: true] nutzt die Action die Antwort (hier, um einen Header hinzuzufügen), lässt aber [NestJS] das senden, was sie zurückgibt. Das ist das Beste aus beiden Welten;
    • Zeilen 38–41: der zu vermeidende Fehler: [@Res()] ohne [passthrough] und ohne Aufrufe von [res.send], [res.redirect], [res.render] .... Der zurückgegebene String wird ignoriert, es wird nie eine Antwort gesendet, und der Browser wartet endlos.

    import { Injectable } from '@nestjs/common';

export interface Product {
  id: number;
  name: string;
  price: number;
}

@Injectable()
export class ProductsService {
  private readonly products: Product[] = [
    { id: 1, name: 'pen', price: 1.5 },
    { id: 2, name: 'notebook', price: 3.2 },
    { id: 3, name: 'eraser', price: 0.8 },
  ];

  findAll(): Product[] {
    return this.products;
  }

  findById(id: number): Product | undefined {
    return this.products.find((p) => p.id === id);
  }
}

    • Zeilen 11–15: die Daten, hier im Speicher;
    • Zeilen 17–23: die beiden Methoden des Services.

    import { Controller, Get, NotFoundException, Param } from '@nestjs/common';
import { Product, ProductsService } from './products.service';

@Controller('products')
export class ProductsController {
  // Dependency Injection: NestJS stellt die Instanz von ProductsService bereit
  constructor(private readonly productsService: ProductsService) {}

  @Get()
  list(): Product[] {
    return this.productsService.findAll();
  }

  @Get(':id')
  oneProduct(@Param('id') id: string): Product {
    const product = this.productsService.findById(Number(id));
    if (!product) {
      throw new NotFoundException(`Product no. ${id} not found`);
    }
    return product;
  }
}
    • Zeile 7: Dependency Injection. Der Controller deklariert in seinem Konstruktor, dass er einen [ProductsService] benötigt; [NestJS] stellt ihm die einzige Instanz des Services bereit. Der Controller ruft nie [new ProductsService()] auf. Die Syntax [private readonly productsService: ProductsService] von [TypeScript] deklariert gleichzeitig den Parameter und das Feld [this.productsService];
    • Zeilen 9–12: Die Action delegiert an den Service;
    • Zeilen 14–21: [Number(id)] wandelt den Routenparameter (einen String) in eine Zahl um. Ein nicht vorhandenes Produkt löst eine 404-Exception aus.

    import { Controller, Get } from '@nestjs/common';
import { ProductsService } from '../products/products.service';

@Controller('clients')
export class ClientsController {
  // ein Service aus einem ANDEREN Modul, das von diesem Modul exportiert wird
  constructor(private readonly productsService: ProductsService) {}

  @Get('cart')
  cart(): object {
    const products = this.productsService.findAll();
    const total = products.reduce((sum, p) => sum + p.price, 0);
    return { client: 'Jean Martin', items: products.length, total };
  }
}
    • Zeile 7: In [ClientsController] wird ein Service aus einem anderen Modul injiziert;
    • Zeilen 9–14: Er verwendet diesen Service, um die Summe eines Warenkorbs zu berechnen.
